SLIDE 5: MAIN CODE EXPLAINED — 1. SMART CONTRACT (ProvenanceRegistry.sol)
================================================================================
[EXPLANATION FOR CLASSMATES]:
"Think of a smart contract as a permanent, read-only database where every student and employer can see the records, but no one can alter or delete existing rows."

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract ProvenanceRegistry {
    // 1. Data structure stored on the blockchain
    struct MediaRecord {
        bytes32 fileHash;    // 32-byte SHA-256 fingerprint of the certificate
        string ipfsCID;      // Permanent IPFS link to the certificate metadata
        address creator;     // Signing wallet of the University Registrar
        uint256 timestamp;   // Immutable block timestamp (proof of date)
        bool isVerified;     // True once successfully registered
    }

    // 2. Mapping: Hash -> Certificate Record
    mapping(bytes32 => MediaRecord) private _records;

    // 3. Register Function: Called by Registrar to anchor certificate
    function registerMedia(bytes32 _fileHash, string calldata _ipfsCID) external {
        if (_records[_fileHash].isVerified) {
            revert MediaAlreadyRegistered(_fileHash, msg.sender, block.timestamp);
        }

        // Save permanently on-chain with msg.sender as creator
        _records[_fileHash] = MediaRecord({
            fileHash: _fileHash,
            ipfsCID: _ipfsCID,
            creator: msg.sender,
            timestamp: block.timestamp,
            isVerified: true
        });

        emit MediaRegistered(_fileHash, _ipfsCID, msg.sender, block.timestamp);
    }

    // 4. Verify Function: Called by employers / foreign universities (Zero gas cost)
    function verifyMedia(bytes32 _fileHash) external view returns (MediaRecord memory) {
        return _records[_fileHash];
    }
}
```

[KEY CONCEPT FOR CLASSMATES]:
- Notice `external view`: Verifying a certificate is 100% FREE! It requires zero gas fees because reading from the blockchain costs nothing. Only registering a new certificate consumes a tiny transaction fee.

SLIDE 6: MAIN CODE EXPLAINED — 2. CLIENT-SIDE CRYPTOGRAPHIC HASHING
================================================================================
[EXPLANATION FOR CLASSMATES]:
"Why don't we upload the whole 20MB degree image directly to the blockchain? Because storing images directly on Ethereum costs thousands of dollars! Instead, we generate a 32-byte SHA-256 fingerprint right inside the student's browser."

```typescript
// Location: frontend/lib/hashing.ts
export async function calculateClientSha256(file: File): Promise<ClientHashResult> {
  // Read file bytes into memory buffer
  const arrayBuffer = await file.arrayBuffer();

  // Use browser's native hardware-accelerated Web Crypto API
  const hashBuffer = await window.crypto.subtle.digest('SHA-256', arrayBuffer);

  // Convert raw 32-byte buffer into 64-character hexadecimal string
  const hashArray = Array.from(new Uint8Array(hashBuffer));
  const rawHash = hashArray.map((b) => b.toString(16).padStart(2, '0')).join('');

  return {
    algorithm: 'SHA-256',
    hash: `0x${rawHash}` as `0x${string}`, // Formatted for Solidity bytes32
    rawHash,
    size: file.size,
  };
}
```

[THE AVALANCHE EFFECT]:
- Authentic Degree (CGPA 7.42) -> `0x4bdd642cef73f5dc4097d62b80d6d91c4d0be312ccc6f7d7...`
- Tampered Degree (Fake 9.85)  -> `0x8a12f904e287bc19a008479784efc830c73de8080b90e24...`
Even changing a single number (7 -> 9) completely randomizes all 64 characters!

async function registerMedia(buffer: Buffer, filename: string, creatorAddress: string) {
  // 1. Calculate authoritative server hash
  const hashResult = hashingService.calculateSha256(buffer);

  // 2. Upload high-res degree image to IPFS
  const mediaCID = await ipfsService.uploadMedia(buffer, filename, 'image/png');

  // 3. Construct provenance metadata manifest
  const metadata = {
    schemaVersion: '1.0',
    fileHash: hashResult.hash,
    fileName: filename,
    createdAt: new Date().toISOString(),
    creator: creatorAddress, // Registrar wallet: 0x3968...7c28
    ipfsCID: mediaCID,
  };

  // 4. Pin metadata bundle to IPFS
  const metadataCID = await ipfsService.uploadMetadata(metadata, `Meta_${filename}`);

  // 5. Submit on-chain anchor transaction to Ethereum
  const tx = await blockchainService.registerMedia(hashResult.hash, metadataCID);

  return { status: 'REGISTERED', txHash: tx.txHash, blockNumber: tx.blockNumber };
}

SLIDE 8: MAIN CODE EXPLAINED — 4. PYTHON PERCEPTUAL FORENSICS (pHash)
================================================================================
[EXPLANATION FOR CLASSMATES]:
"What if someone doesn't forge the grade, but just re-saves the JPEG on WhatsApp or compresses it? The SHA-256 hash would change. That's why we built a Python Perceptual Hashing (DCT) engine to distinguish between harmless compression vs. malicious forgery!"

```python
# Location: forensics/analyze.py
import imagehash
from PIL import Image

def analyze_tampering(original_path, suspect_path):
    # Calculate Discrete Cosine Transform (DCT) Perceptual Hash
    phash_orig = imagehash.phash(Image.open(original_path))
    phash_susp = imagehash.phash(Image.open(suspect_path))

    # Calculate Hamming Distance (number of differing bits out of 64)
    hamming_distance = phash_orig - phash_susp

    if hamming_distance == 0:
        return "IDENTICAL: No visual changes detected."
    elif hamming_distance <= 10:
        return "COMPRESSED/RESIZED: Visual structure intact (benign modification)."
    else:
        return "TAMPERED / DEEPFAKE: Visual geometry altered!"
```
